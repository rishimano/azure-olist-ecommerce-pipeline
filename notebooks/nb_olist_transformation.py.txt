# Databricks notebook source
# Project: End-to-End Azure E-Commerce Analytics Pipeline
# Dataset: Olist Brazilian E-Commerce Dataset
# Tools: ADLS Gen2, Databricks, PySpark, Azure SQL, ADF, Power BI

# COMMAND ----------

from pyspark.sql.functions import (
    avg,
    col,
    countDistinct,
    datediff,
    month,
    round as spark_round,
    sum as spark_sum,
    to_date,
    year,
)

# COMMAND ----------

# Configuration
# Keep real credentials out of this notebook and GitHub.
# Replace only the non-secret resource-name placeholders below.

storage_account = "YOUR_STORAGE_ACCOUNT_NAME"
container_raw = "raw"
container_curated = "curated"

sql_server = "YOUR_SQL_SERVER_NAME.database.windows.net"
sql_database = "YOUR_SQL_DATABASE_NAME"
sql_username = "YOUR_SQL_USERNAME"

# Secrets should be stored in a Databricks secret scope.
secret_scope = "YOUR_SECRET_SCOPE"
storage_key_secret = "YOUR_STORAGE_KEY_SECRET_NAME"
sql_password_secret = "YOUR_SQL_PASSWORD_SECRET_NAME"

storage_key = dbutils.secrets.get(
    scope=secret_scope,
    key=storage_key_secret,
)

sql_password = dbutils.secrets.get(
    scope=secret_scope,
    key=sql_password_secret,
)

spark.conf.set(
    f"fs.azure.account.key.{storage_account}.dfs.core.windows.net",
    storage_key,
)

raw_path = (
    f"abfss://{container_raw}@"
    f"{storage_account}.dfs.core.windows.net/"
)

curated_path = (
    f"abfss://{container_curated}@"
    f"{storage_account}.dfs.core.windows.net/"
)

# COMMAND ----------

# Load raw CSV files from ADLS Gen2

orders_df = spark.read.csv(
    f"{raw_path}olist_orders_dataset.csv",
    header=True,
    inferSchema=True,
)

order_items_df = spark.read.csv(
    f"{raw_path}olist_order_items_dataset.csv",
    header=True,
    inferSchema=True,
)

customers_df = spark.read.csv(
    f"{raw_path}olist_customers_dataset.csv",
    header=True,
    inferSchema=True,
)

products_df = spark.read.csv(
    f"{raw_path}olist_products_dataset.csv",
    header=True,
    inferSchema=True,
)

payments_df = spark.read.csv(
    f"{raw_path}olist_order_payments_dataset.csv",
    header=True,
    inferSchema=True,
)

reviews_df = spark.read.csv(
    f"{raw_path}olist_order_reviews_dataset.csv",
    header=True,
    inferSchema=True,
)

sellers_df = spark.read.csv(
    f"{raw_path}olist_sellers_dataset.csv",
    header=True,
    inferSchema=True,
)

category_translation_df = spark.read.csv(
    f"{raw_path}product_category_name_translation.csv",
    header=True,
    inferSchema=True,
)

print("Raw datasets loaded successfully.")

# COMMAND ----------

# Validate raw row counts

raw_counts = {
    "orders": orders_df.count(),
    "order_items": order_items_df.count(),
    "customers": customers_df.count(),
    "products": products_df.count(),
    "payments": payments_df.count(),
    "reviews": reviews_df.count(),
    "sellers": sellers_df.count(),
    "category_translation": category_translation_df.count(),
}

print("Raw dataset row counts:")

for dataset_name, row_count in raw_counts.items():
    print(f"{dataset_name}: {row_count:,} rows")

# COMMAND ----------

# Clean orders

orders_clean_df = (
    orders_df
    .dropDuplicates(["order_id"])
    .withColumn(
        "order_purchase_date",
        to_date(col("order_purchase_timestamp")),
    )
    .withColumn(
        "order_delivered_date",
        to_date(col("order_delivered_customer_date")),
    )
    .withColumn(
        "order_estimated_delivery_date",
        to_date(col("order_estimated_delivery_date")),
    )
    .withColumn(
        "delivery_days",
        datediff(
            col("order_delivered_date"),
            col("order_purchase_date"),
        ),
    )
    .withColumn(
        "late_delivery_days",
        datediff(
            col("order_delivered_date"),
            col("order_estimated_delivery_date"),
        ),
    )
)

# COMMAND ----------

# Clean supporting datasets

order_items_clean_df = (
    order_items_df
    .dropDuplicates()
    .withColumn(
        "price",
        col("price").cast("double"),
    )
    .withColumn(
        "freight_value",
        col("freight_value").cast("double"),
    )
)

customers_clean_df = customers_df.dropDuplicates(
    ["customer_id"]
)

products_clean_df = products_df.dropDuplicates(
    ["product_id"]
)

sellers_clean_df = sellers_df.dropDuplicates(
    ["seller_id"]
)

payments_clean_df = (
    payments_df
    .dropDuplicates()
    .withColumn(
        "payment_value",
        col("payment_value").cast("double"),
    )
    .withColumn(
        "payment_installments",
        col("payment_installments").cast("int"),
    )
)

reviews_clean_df = (
    reviews_df
    .dropDuplicates(["review_id"])
    .withColumn(
        "review_score",
        col("review_score").cast("int"),
    )
)

category_translation_clean_df = (
    category_translation_df
    .dropDuplicates(["product_category_name"])
)

# COMMAND ----------

# Validate cleaned row counts

clean_counts = {
    "orders": orders_clean_df.count(),
    "order_items": order_items_clean_df.count(),
    "customers": customers_clean_df.count(),
    "products": products_clean_df.count(),
    "payments": payments_clean_df.count(),
    "reviews": reviews_clean_df.count(),
    "sellers": sellers_clean_df.count(),
    "category_translation": category_translation_clean_df.count(),
}

print("Cleaned dataset row counts:")

for dataset_name, row_count in clean_counts.items():
    print(f"{dataset_name}: {row_count:,} rows")

# COMMAND ----------

# Enrich products with English category names

products_enriched_df = products_clean_df.join(
    category_translation_clean_df,
    on="product_category_name",
    how="left",
)

# COMMAND ----------

# Build fact table at order-item grain
#
# Payments and reviews are cleaned and validated separately.
# They are not directly joined to the order-item fact table because
# both datasets may contain multiple records per order. Joining them
# without prior aggregation could duplicate fact rows.

fact_orders_df = (
    order_items_clean_df
    .join(
        orders_clean_df,
        on="order_id",
        how="left",
    )
    .join(
        customers_clean_df,
        on="customer_id",
        how="left",
    )
    .join(
        products_enriched_df,
        on="product_id",
        how="left",
    )
    .join(
        sellers_clean_df,
        on="seller_id",
        how="left",
    )
)

fact_orders_final_df = fact_orders_df.select(
    "order_id",
    "order_item_id",
    "customer_id",
    "seller_id",
    "product_id",
    "order_status",
    "order_purchase_date",
    "order_delivered_date",
    "order_estimated_delivery_date",
    "delivery_days",
    "late_delivery_days",
    "customer_city",
    "customer_state",
    "seller_city",
    "seller_state",
    "product_category_name",
    "product_category_name_english",
    "price",
    "freight_value",
)

fact_row_count = fact_orders_final_df.count()

print(f"fact_orders: {fact_row_count:,} rows")

# COMMAND ----------

# Create monthly sales summary

monthly_sales_summary_df = (
    fact_orders_final_df
    .groupBy(
        year("order_purchase_date").alias("sales_year"),
        month("order_purchase_date").alias("sales_month"),
    )
    .agg(
        spark_round(
            spark_sum("price"),
            2,
        ).alias("total_revenue"),
        countDistinct(
            "order_id"
        ).alias("total_orders"),
        spark_round(
            avg("price"),
            2,
        ).alias("average_item_value"),
    )
    .orderBy(
        "sales_year",
        "sales_month",
    )
)

# COMMAND ----------

# Create sales-by-state summary

sales_by_state_df = (
    fact_orders_final_df
    .groupBy("customer_state")
    .agg(
        spark_round(
            spark_sum("price"),
            2,
        ).alias("total_revenue"),
        countDistinct(
            "order_id"
        ).alias("total_orders"),
        spark_round(
            avg("price"),
            2,
        ).alias("average_item_value"),
    )
    .orderBy(
        col("total_revenue").desc()
    )
)

# COMMAND ----------

# Create product-category performance summary

category_performance_df = (
    fact_orders_final_df
    .groupBy("product_category_name_english")
    .agg(
        spark_round(
            spark_sum("price"),
            2,
        ).alias("total_revenue"),
        countDistinct(
            "order_id"
        ).alias("total_orders"),
        spark_round(
            avg("price"),
            2,
        ).alias("average_item_value"),
        spark_round(
            avg("delivery_days"),
            2,
        ).alias("average_delivery_days"),
    )
    .orderBy(
        col("total_revenue").desc()
    )
)

# COMMAND ----------

# Create delivery-performance summary

delivery_performance_df = (
    fact_orders_final_df
    .groupBy("customer_state")
    .agg(
        spark_round(
            avg("delivery_days"),
            2,
        ).alias("average_delivery_days"),
        spark_round(
            avg("late_delivery_days"),
            2,
        ).alias("average_late_delivery_days"),
        countDistinct(
            "order_id"
        ).alias("total_orders"),
    )
    .orderBy(
        col("average_delivery_days").desc()
    )
)

# COMMAND ----------

# Save curated outputs to ADLS Gen2 in Parquet format

curated_tables = {
    "fact_orders": fact_orders_final_df,
    "monthly_sales_summary": monthly_sales_summary_df,
    "sales_by_state": sales_by_state_df,
    "category_performance": category_performance_df,
    "delivery_performance": delivery_performance_df,
}

for table_name, dataframe in curated_tables.items():
    output_path = f"{curated_path}{table_name}"

    dataframe.write \
        .mode("overwrite") \
        .parquet(output_path)

    print(
        f"Saved {table_name} to {output_path}"
    )

# COMMAND ----------

# Configure Azure SQL Database JDBC connection

jdbc_url = (
    f"jdbc:sqlserver://{sql_server}:1433;"
    f"database={sql_database};"
    "encrypt=true;"
    "trustServerCertificate=false;"
    "hostNameInCertificate=*.database.windows.net;"
    "loginTimeout=30;"
)

connection_properties = {
    "user": sql_username,
    "password": sql_password,
    "driver": "com.microsoft.sqlserver.jdbc.SQLServerDriver",
}

# COMMAND ----------

# Test Azure SQL connectivity before writing tables

connection_test_df = spark.read.jdbc(
    url=jdbc_url,
    table="(SELECT 1 AS connection_test) AS test_table",
    properties=connection_properties,
)

connection_test_df.show()

print("Azure SQL connection test succeeded.")

# COMMAND ----------

# Write curated tables to Azure SQL Database

for table_name, dataframe in curated_tables.items():

    dataframe.write.jdbc(
        url=jdbc_url,
        table=table_name,
        mode="overwrite",
        properties=connection_properties,
    )

    print(
        f"Wrote {table_name} to Azure SQL Database."
    )

# COMMAND ----------

# Verify final Azure SQL row counts

print("Azure SQL table verification:")

for table_name in curated_tables:

    sql_df = spark.read.jdbc(
        url=jdbc_url,
        table=table_name,
        properties=connection_properties,
    )

    print(
        f"{table_name}: {sql_df.count():,} rows"
    )

# COMMAND ----------

print("Pipeline completed successfully.")
